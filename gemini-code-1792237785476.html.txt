<script>
document.addEventListener('DOMContentLoaded', () => {
const $=id=>document.getElementById(id), view=$('view'), vctx=view.getContext('2d');
let W=900, H=1200, tool='brush', drawing=false, start=null, last=null, preview=null, history=[], future=[], active=1, layerSeq=2, showGrid=false, flip=false, hand=false, panStart=null, offset={x:0,y:0};
let layers=[];

function makeLayer(name,locked=false){
  let c=document.createElement('canvas');
  c.width=W; c.height=H;
  return {id:layerSeq++, name, canvas:c, opacity:1, blend:'source-over', visible:true, locked, transform:{x:0, y:0, scale:1, rotate:0}};
}

function init(){
  layers=[];
  let bg=makeLayer('背景',true);
  bg.canvas.getContext('2d').fillStyle='#ffffff';
  bg.canvas.getContext('2d').fillRect(0,0,W,H);
  layers.push(bg);
  layers.push(makeLayer('レイヤー 1'));
  active=1; history=[]; future=[];
  snapshot(); renderLayers(); render(); resizePaper();
}

function current(){return layers[active]}
function ctx(layer=current()){return layer.canvas.getContext('2d')}

function snapshot(){
  try{
    history.push(layers.map(l=>({
      ...l,
      canvas:cloneCanvas(l.canvas),
      transform:{...l.transform}
    })));
    if(history.length>35)history.shift();
    future=[];
  }catch(e){}
}

function cloneCanvas(c){
  let n=document.createElement('canvas');
  n.width=c.width; n.height=c.height;
  n.getContext('2d').drawImage(c,0,0);
  return n;
}

function restore(arr){
  layers=arr.map(l=>({
    ...l,
    canvas:cloneCanvas(l.canvas),
    transform:{...(l.transform||{x:0,y:0,scale:1,rotate:0})}
  }));
  active=Math.min(active,layers.length-1);
  renderLayers(); render();
}

function undo(){
  if(history.length<2)return;
  future.push(layers.map(l=>({...l, canvas:cloneCanvas(l.canvas), transform:{...l.transform}})));
  history.pop();
  restore(history[history.length-1]);
}

function redo(){
  if(!future.length)return;
  history.push(layers.map(l=>({...l, canvas:cloneCanvas(l.canvas), transform:{...l.transform}})));
  restore(future.pop());
}

function render(){
  vctx.clearRect(0,0,W,H);
  for(const l of layers){
    if(!l.visible)continue;
    vctx.save();
    vctx.globalAlpha=l.opacity;
    vctx.globalCompositeOperation=l.blend;
    if(l.transform && (l.transform.scale!==1 || l.transform.rotate!==0 || l.transform.x!==0 || l.transform.y!==0)){
      vctx.translate(W/2 + l.transform.x, H/2 + l.transform.y);
      vctx.rotate(l.transform.rotate * Math.PI / 180);
      vctx.scale(l.transform.scale, l.transform.scale);
      vctx.drawImage(l.canvas, -W/2, -H/2);
    } else {
      vctx.drawImage(l.canvas,0,0);
    }
    vctx.restore();
  }
  if(preview){
    vctx.save();
    vctx.globalAlpha=+$('opacity').value/100;
    vctx.strokeStyle=$('color').value;
    vctx.lineWidth=+$('size').value;
    vctx.lineCap='round';
    vctx.beginPath();
    if(tool==='line'){
      vctx.moveTo(preview.x1,preview.y1);
      vctx.lineTo(preview.x2,preview.y2);
      vctx.stroke();
    }else if(tool==='rect'){
      vctx.strokeRect(preview.x1,preview.y1,preview.x2-preview.x1,preview.y2-preview.y1);
    }else if(tool==='ellipse'){
      vctx.beginPath();
      vctx.ellipse((preview.x1+preview.x2)/2,(preview.y1+preview.y2)/2,Math.abs(preview.x2-preview.x1)/2,Math.abs(preview.y2-preview.y1)/2,0,0,Math.PI*2);
      vctx.stroke();
    }
    vctx.restore();
  }
  if(showGrid){
    vctx.save();
    vctx.strokeStyle='#77777755';
    vctx.lineWidth=1;
    for(let x=0;x<W;x+=50){vctx.beginPath();vctx.moveTo(x,0);vctx.lineTo(x,H);vctx.stroke()}
    for(let y=0;y<H;y+=50){vctx.beginPath();vctx.moveTo(0,y);vctx.lineTo(W,y);vctx.stroke()}
    vctx.restore();
  }
  if(flip){
    vctx.save();
    vctx.translate(W,0);
    vctx.scale(-1,1);
    let temp=cloneCanvas(view);
    vctx.clearRect(0,0,W,H);
    vctx.drawImage(temp,0,0);
    vctx.restore();
  }
  $('docSize').textContent=`${W} × ${H} px`;
  $('layerCount').textContent=layers.length;
}

function resizePaper(){
  let ratio=W/H, stage=$('stage'), maxW=stage.clientWidth*.92, maxH=stage.clientHeight*.91, w=Math.min(maxW,maxH*ratio), h=w/ratio;
  $('paperWrap').style.width=w+'px';
  $('paperWrap').style.height=h+'px';$('zoomLabel').textContent=Math.round(w/W*100)+'%';
}

function renderLayers(){
  let box=$('layers');
  box.innerHTML='';
  layers.forEach((l,i)=>{
    let el=document.createElement('div');
    el.className='layer'+(i===active?' selected':'');
    el.draggable=!l.locked;
    el.innerHTML=`<span>${l.visible?'◉':'○'}</span><div><input class="layerName" value="${escapeHtml(l.name)}" aria-label="レイヤー名"><small>${l.locked?'ロック中':'ラスターレイヤー'}</small></div><button title="選択">⋯</button>`;
    el.addEventListener('click',e=>{if(e.target.tagName==='INPUT')return;active=i;renderLayers();syncLayerControls()});
    el.querySelector('input').addEventListener('change',e=>{l.name=e.target.value;renderLayers()});
    el.addEventListener('dragstart',e=>{e.dataTransfer.setData('text/plain',i)});
    el.addEventListener('dragover',e=>e.preventDefault());
    el.addEventListener('drop',e=>{
      e.preventDefault();
      let from=+e.dataTransfer.getData('text/plain'), to=i;
      if(from===0||to===0||from===to)return;
      let item=layers.splice(from,1)[0];
      layers.splice(to,0,item);
      active=to; snapshot(); renderLayers(); render();
    });
    box.appendChild(el);
  });
  syncLayerControls();
}

function escapeHtml(s){return s.replace(/[&<>"']/g,c=>({'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;',"'":'&#39;'}[c]))}

function syncLayerControls(){
  let l=current();
  if(!l)return;
  $('layerOpacity').value=Math.round(l.opacity*100);
  $('layerOpacityVal').textContent=Math.round(l.opacity*100)+'\%';$('blend').value=l.blend;
  $('visible').checked=l.visible;
  $('locked').checked=l.locked;
  if(!l.transform) l.transform={x:0,y:0,scale:1,rotate:0};
  $('imgScale').value=Math.round(l.transform.scale*100);
  $('imgScaleVal').textContent=Math.round(l.transform.scale*100)+'\%';$('imgRotate').value=l.transform.rotate;
  $('imgRotateVal').textContent=l.transform.rotate+'°';$('imgPosX').value=l.transform.x;
  $('imgPosY').value=l.transform.y;
}

function point(e){
  let r=view.getBoundingClientRect();
  return {x:(e.clientX-r.left)/r.width*W, y:(e.clientY-r.top)/r.height*H};
}

function setupStroke(c){
  c.lineCap='round';
  c.lineJoin='round';
  c.lineWidth=+$('size').value;
  c.globalAlpha=+$('opacity').value/100;
  c.globalCompositeOperation=tool==='eraser'?'destination-out':'source-over';
  c.strokeStyle=$('color').value;
  c.fillStyle=$('color').value;
  if($('brushType').value==='square')c.lineCap='butt';
}

function dot(p){
  let c=ctx();
  setupStroke(c);
  c.beginPath();
  c.arc(p.x,p.y,Math.max(0.5,+$('size').value/2),0,Math.PI*2);
  c.fill();
}

function stroke(a,b){
  let c=ctx();
  setupStroke(c);
  c.beginPath();
  c.moveTo(a.x,a.y);
  let smooth=+$('stabilizer').value/20;
  if(smooth>0){
    let p={x:a.x*(1-smooth)+b.x*smooth, y:a.y*(1-smooth)+b.y*smooth};
    c.quadraticCurveTo(a.x,a.y,p.x,p.y);
  }else c.lineTo(b.x,b.y);
  c.stroke();
}

function pointerDown(e){
  if(hand){
    panStart={x:e.clientX, y:e.clientY, ox:offset.x, oy:offset.y};
    return;
  }
  if(!current()||current().locked)return;
  let p=point(e);
  view.setPointerCapture(e.pointerId);
  if(tool==='eyedrop'){
    let c=vctx.getImageData(Math.round(p.x
